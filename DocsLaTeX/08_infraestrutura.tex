\section{Introdução}
Este documento descreve a infraestrutura necessária contemplando os requisitos de hardware, software e serviços exigidos para que o sistema Kanban seja posto em produção.

\section{Requisitos mínimos - Hardware}
% [Hardware necessario à implantação do sistema]
\begin{itemize}
    \item \textbf{Servidor de Aplicação:} 
    \begin{itemize}
        \item Processador: Mínimo 2 vCPUs (ex: AWS EC2 t3.micro ou equivalente).
        \item Memória RAM: Mínimo 2 GB.
        \item Armazenamento: 20 GB SSD.
    \end{itemize}
    \item \textbf{Servidor de Banco de Dados:} 
    \begin{itemize}
        \item Processador: 2 vCPUs.
        \item Memória RAM: 4 GB.
        \item Armazenamento: 50 GB.
    \end{itemize}
    \item \textbf{Cliente (Usuário Final) - Computador}
    \begin{itemize}
        \item Processador: Pentium 4 ou superior compatível com SSE3.
        \item Armazenamento: 350 MB.
        \item Memória RAM: 4GB.
    \end{itemize}
    \item \textbf{Cliente (Usuário Final) - Dispositivos móveis}
    \begin{itemize}
        \item Armazenamento: 32 GB no Android ou 200 MB de espaço livre no iOS.
        \item Memória RAM: 4GB.
    \end{itemize}
\end{itemize}

\section{Requisitos mínimos - Software}
% [Software necessario à implantação do sistema]
\begin{itemize}
    \item \textbf{Sistema Operacional do Servidor:} Windows 7 ou superior, Linux (ex: Ubuntu Server 22.04 LTS).
    \item \textbf{Ambiente de Execução (Backend):} Python 3.11 ou superior.
    \item \textbf{Banco de Dados:}  MySQL com MySQL Workbench.
    \item \textbf{Cliente - Windows:} Windows 10 ou superior / Windows Server 2016 ou superior.
    \item \textbf{Cliente - Linux:} Ubuntu 18.04+ de 64 bits, Debian 10+, openSUSE 15.5+ ou Fedora Linux 39+.
    \item \textbf{Cliente - Mac:} macOS 12 Monterey e versões mais recentes.
    \item \textbf{Cliente - Android:} Android 10 ou superior.
    \item \textbf{Cliente - iOS:} iOS 17 ou superior.
\end{itemize}

\section{Serviços necessários à implantação do sistema em produção}
% [Serviços necessarios à implantação do sistema]
\begin{itemize}
    \item \textbf{Hospedagem da Aplicação em Nuvem (Cloud Hosting):} Streamlit Community Cloud para hospedagem gratuita da ferramenta.
    \item \textbf{Hospedagem do BD em Nuvem:} Google Cloud SQL Database, para hospedar a base MySQL, por exemplo.
\end{itemize}
